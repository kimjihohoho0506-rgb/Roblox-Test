```local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer

local CHECK_INTERVAL = 0.05
local lastSafeCFrame = nil
local timePassed = 0
local isTPing = false

if getgenv().AntiFlingRollback then
    getgenv().AntiFlingRollback:Disconnect()
end

getgenv().AntiFlingRollback = RunService.Heartbeat:Connect(function(dt)
    local character = LocalPlayer.Character
    if not character then return end
    
    local hrp = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    
    if not hrp or not humanoid or humanoid.Health <= 0 then return end

    if humanoid.FloorMaterial ~= Enum.Material.Air then
        timePassed = timePassed + dt
        if timePassed >= CHECK_INTERVAL then
            timePassed = 0
            lastSafeCFrame = hrp.CFrame
        end
    else
        if not isTPing and lastSafeCFrame then
            isTPing = true
            local targetCFrame = lastSafeCFrame + Vector3.new(0, 1, 0)

            task.spawn(function()
                for _ = 1, 5 do
                    if not hrp or not character then break end
                    
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                    
                    hrp.CFrame = targetCFrame
                    character:PivotTo(targetCFrame)
                    
                    RunService.RenderStepped:Wait()
                end
                
                task.wait(0.05)
                isTPing = false
            end)
        end
    end
end)
```
